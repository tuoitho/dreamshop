# Cart Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an authenticated, server-side cart that validates product availability and soft-deletion status, and provides cart and order-preview UI flows.

**Architecture:** A new Go `cart-service` owns persisted cart lines in `cart_db`. It enriches cart lines through direct gRPC calls to Catalog for product truth and Inventory for availability. BFF adapts CartService to authenticated REST endpoints behind Kong; the React client keeps only line selection locally and renders the validated cart/order preview.

**Tech Stack:** Go 1.24, PostgreSQL 16/pgx, gRPC/Protocol Buffers, Kong, React 19, TypeScript, Vite, Vitest, Testing Library.

## Global Constraints

- Preserve bounded-context ownership: Cart owns cart rows; Catalog owns product data and deleted state; Inventory owns stock and availability.
- Internal service calls use direct private gRPC only. Browser requests go through Kong and BFF.
- Cart endpoints require Kong `caller-context`; BFF trusts `X-User-Id` only on this authenticated edge path.
- Do not manually edit generated files under `contracts/go/` or `contracts/java/src/main/generated/`.
- Contract changes use `buf format --diff --exit-code`, `buf lint`, and `buf generate`; commit generated Go and Java output with proto sources.
- Run the generated-output `git diff --exit-code` gate only in a clean feature worktree containing the intended contract change. The current workspace has unrelated dirty generated files, so use `buf generate` plus targeted diff inspection until that baseline is isolated.
- All REST responses use `{ "data": ..., "message": "..." }`; errors use stable `code`, client-safe `message`, optional `errors`, and `traceId`.
- Cart selection is client-side state. The persisted cart has no `selected` column.
- A `LIMITED` line is not silently clamped; it returns `availableQuantity` and the UI lets the customer adopt it.
- The current worktree contains unrelated uncommitted files. Never use `git add -A`, `git commit -a`, or a broad path. Every commit command must list only files changed by its task.

```proto
syntax = "proto3";

package ecommerce.cart.v1;

import "common/v1/common.proto";

option go_package = "ecommerce.local/contracts/go/cart/v1;cartv1";
option java_multiple_files = true;
option java_package = "com.ecommerce.contracts.cart.v1";

service CartService {
  rpc GetCart(GetCartRequest) returns (GetCartResponse);
  rpc AddItem(AddItemRequest) returns (AddItemResponse);
  rpc UpdateItemQuantity(UpdateItemQuantityRequest) returns (UpdateItemQuantityResponse);
  rpc RemoveItem(RemoveItemRequest) returns (RemoveItemResponse);
  rpc ClearCart(ClearCartRequest) returns (ClearCartResponse);
}

message GetCartRequest {
  string user_id = 1;
}

message AddItemRequest {
  string user_id = 1;
  string product_id = 2;
  string variant_label = 3;
  int32 quantity = 4;
}

message UpdateItemQuantityRequest {
  string user_id = 1;
  string item_id = 2;
  int32 quantity = 3;
}

message RemoveItemRequest {
  string user_id = 1;
  string item_id = 2;
}

message ClearCartRequest {
  string user_id = 1;
}

message Cart {
  repeated CartItem items = 1;
  bool degraded = 2;
}

message GetCartResponse {
  Cart cart = 1;
}

message AddItemResponse {
  Cart cart = 1;
}

message UpdateItemQuantityResponse {
  Cart cart = 1;
}

message RemoveItemResponse {
  Cart cart = 1;
}

message ClearCartResponse {
  Cart cart = 1;
}

message CartItem {
  string item_id = 1;
  string product_id = 2;
  string variant_label = 3;
  int32 quantity = 4;
  string name = 5;
  string brand = 6;
  string category_name = 7;
  ecommerce.common.v1.Money unit_price = 8;
  string image_emoji = 9;
  repeated uint32 image_hue = 10;
  string image_url = 11;
  string variant_color_hex = 12;
  CartItemStatus status = 13;
  int32 available_quantity = 14;
}

enum CartItemStatus {
  CART_ITEM_STATUS_UNSPECIFIED = 0;
  CART_ITEM_STATUS_AVAILABLE = 1;
  CART_ITEM_STATUS_LIMITED = 2;
  CART_ITEM_STATUS_OUT_OF_STOCK = 3;
  CART_ITEM_STATUS_UNAVAILABLE = 4;
}
```

Each RPC returns the response type named for that RPC; every response wraps a
`Cart` so mutation callers receive the validated cart in one round trip. The
BFF maps the prefixed `CART_ITEM_STATUS_*` enum names to the public REST status
strings `AVAILABLE`, `LIMITED`, `OUT_OF_STOCK`, and `UNAVAILABLE`; the React
types and visible behavior use those unprefixed strings.

```go
func (s *Server) SoftDeleteProduct(ctx context.Context, req *catalogv1.SoftDeleteProductRequest) (*catalogv1.SoftDeleteProductResponse, error) {
	id := strings.TrimSpace(req.GetId())
	if id == "" {
		return nil, status.Error(codes.InvalidArgument, "id is required")
	}
	if err := s.repo.SoftDelete(ctx, id, req.GetDeleted()); err != nil {
		if errors.Is(err, domain.ErrProductNotFound) {
			return nil, notFoundError(id)
		}
		return nil, status.Errorf(codes.Internal, "set product deleted %q: %v", id, err)
	}
	return &catalogv1.SoftDeleteProductResponse{Id: id, Deleted: req.GetDeleted()}, nil
}
```

```go
type callerKey struct{}

func RequireCaller(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		userID := strings.TrimSpace(r.Header.Get("X-User-Id"))
		if userID == "" {
			writeError(w, r, http.StatusUnauthorized, "UNAUTHORIZED", "Authentication is required.", nil)
			return
		}
		next.ServeHTTP(w, r.WithContext(context.WithValue(r.Context(), callerKey{}, userID)))
	})
}

func callerID(r *http.Request) string {
	value, _ := r.Context().Value(callerKey{}).(string)
	return value
}
```
